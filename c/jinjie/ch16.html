第 16 章 综合案例：学生成绩管理

预计 10 分钟·约 1,774 字·C 语言进阶

把前三部分的知识合起来做一个完整的小程序：读入学生信息、求平均分、按成绩排序、按学号查找。这个程序覆盖了结构体、结构体数组、指针、qsort、字符串，是期末大作业的雏形。

C
#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAXN 100

typedef struct {
    int    id;
    char   name[20];
    double score;
} Student;

/* 输入 n 个学生 */
void input(Student a[], int n) {
    for (int i = 0; i < n; i++) {
        scanf("%d %19s %lf", &a[i].id, a[i].name, &a[i].score);
    }
}

/* 求平均分，只读，所以用 const */
double average(const Student a[], int n) {
    double sum = 0.0;
    for (int i = 0; i < n; i++) {
        sum += a[i].score;
    }
    return n > 0 ? sum / n : 0.0;
}

/* qsort 比较函数：按成绩降序，成绩相同按学号升序 */
int cmp_score(const void *p, const void *q) {
    const Student *a = (const Student *)p;
    const Student *b = (const Student *)q;

    if (a->score < b->score) return 1;
    if (a->score > b->score) return -1;
    return a->id - b->id;
}

/* 按学号查找，返回下标，找不到返回 -1 */
int find_by_id(const Student a[], int n, int id) {
    for (int i = 0; i < n; i++) {
        if (a[i].id == id) return i;
    }
    return -1;
}

/* 输出表格 */
void print_all(const Student a[], int n) {
    printf("\n%-8s %-14s %8s\n", "学号", "姓名", "成绩");
    printf("--------------------------------\n");
    for (int i = 0; i < n; i++) {
        printf("%-8d %-14s %8.1f\n", a[i].id, a[i].name, a[i].score);
    }
}

int main(void) {
    Student list[MAXN];
    int n;

    printf("学生人数: ");
    if (scanf("%d", &n) != 1 || n <= 0 || n > MAXN) {
        printf("人数不合法\n");
        return 1;
    }

    printf("依次输入 学号 姓名 成绩:\n");
    input(list, n);

    printf("\n平均分: %.2f\n", average(list, n));

    qsort(list, n, sizeof(list[0]), cmp_score);
    printf("\n按成绩降序排列:");
    print_all(list, n);

    int id;
    printf("\n输入要查找的学号: ");
    if (scanf("%d", &id) == 1) {
        int idx = find_by_id(list, n, id);
        if (idx >= 0) {
            printf("找到: %d %s %.1f\n", list[idx].id, list[idx].name, list[idx].score);
        } else {
            printf("没有学号为 %d 的学生\n", id);
        }
    }

    return 0;
}

这个程序里值得琢磨的几个点：

位置用到的知识说明
typedef struct第 14 章省掉 struct 关键字
const Student a[]第 4、13 章只读参数，承诺不修改
qsort + 比较函数第 5、8、13 章函数指针、void*、结构体排序
a->score第 12 章结构体指针成员访问
%-8d 等格式格式控制左对齐、定宽，输出成表格
scanf 返回值检查健壮性输入非法时不崩溃
■自己动手

把这个程序扩展成完整的管理系统：增加「插入学生」「删除学生」「修改成绩」「按姓名查找」「统计各分数段人数」等功能。做完它，你的 C 语言课程设计基本就有着落了。
