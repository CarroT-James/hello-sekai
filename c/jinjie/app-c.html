1-4 删除指定元素（双指针法）#

C
int k = 0;                        // k 是"写指针"
for (int i = 0; i < n; i++) {     // i 是"读指针"
    if (a[i] != x) {
        a[k++] = a[i];            // 保留，写到前面
    }
}
n = k;                            // 新长度

1-8 二分查找插入位置#

C
int lo = 0, hi = n;               // 注意 hi = n，不是 n - 1
while (lo < hi) {
    int mid = lo + (hi - lo) / 2;
    if (a[mid] < target) lo = mid + 1;
    else                 hi = mid;
}
return lo;                        // 循环结束时 lo 就是插入位置

1-11 数组去重#

C
int k = 0;
for (int i = 0; i < n; i++) {
    int seen = 0;
    for (int j = 0; j < k; j++) {     // 在已保留的部分里找
        if (a[j] == a[i]) { seen = 1; break; }
    }
    if (!seen) a[k++] = a[i];
}
n = k;

1-12 统计字符#

C
int upper = 0, lower = 0, digit = 0, space = 0, other = 0;
for (int i = 0; s[i] != '\0'; i++) {
    char c = s[i];
    if (c >= 'A' && c <= 'Z')      upper++;
    else if (c >= 'a' && c <= 'z') lower++;
    else if (c >= '0' && c <= '9') digit++;
    else if (c == ' ')             space++;
    else                           other++;
}

习题二#

2-4 手写 strlen#

C
size_t my_strlen(const char *s) {
    const char *p = s;
    while (*p) p++;               // *p 等价于 *p != '\0'
    return (size_t)(p - s);
}

2-7 字符串原地反转#

C
void reverse_str(char *s) {
    char *left = s;
    char *right = s;
    while (*right) right++;       // 走到 '\0'
    right--;                      // 回退到最后一个字符
    while (left < right) {
        char t = *left; *left = *right; *right = t;
        left++; right--;
    }
}

2-12 找 bug#

C
char *s = malloc(10);
s = "hello";        // 错误 1：把指针指向了字符串常量，malloc 的内存泄漏了
free(s);            // 错误 2：free 的是字符串常量的地址，不是 malloc 的地址
                    //         → 未定义行为，通常崩溃

// 正确写法
char *s = malloc(10);
if (s) {
    strcpy(s, "hello");
    printf("%s\n", s);
    free(s);
    s = NULL;
}

3-9 链表基本操作#

C
struct Node { int data; struct Node *next; };

// 头插
struct Node *push_front(struct Node *head, int v) {
    struct Node *p = malloc(sizeof(struct Node));
    if (!p) return head;
    p->data = v;
    p->next = head;
    return p;
}

// 查找
struct Node *find(struct Node *head, int v) {
    for (struct Node *p = head; p; p = p->next)
        if (p->data == v) return p;
    return NULL;
}

// 释放
void destroy(struct Node *head) {
    while (head) {
        struct Node *next = head->next;
        free(head);
        head = next;
    }
}
