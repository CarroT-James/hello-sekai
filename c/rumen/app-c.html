C

a = b;

1-5 温度转换#

C
double c, f;
scanf("%lf", &c);
f = c * 9.0 / 5.0 + 32;       // 注意写 9.0 而不是 9
printf("%.1f\n", f);

第 2 章#

2-4 三位数反转#

C
int n, a, b, c;
scanf("%d", &n);
a = n / 100;          // 百位
b = n / 10 % 10;      // 十位
c = n % 10;           // 个位
printf("%d%d%d\n", c, b, a);

2-5 成绩等级#

C
int score;
scanf("%d", &score);

if (score >= 90)      printf("A\n");
else if (score >= 80) printf("B\n");
else if (score >= 70) printf("C\n");
else if (score >= 60) printf("D\n");
else                  printf("E\n");

第 3 章#

3-1 判断闰年#

C
int y;
scanf("%d", &y);
if ((y % 4 == 0 && y % 100 != 0) || y % 400 == 0)
    printf("闰年\n");
else
    printf("平年\n");

3-3 求和#

C
int n, sum = 0;
scanf("%d", &n);
for (int i = 1; i <= n; i++) {
    sum += i;
}
printf("%d\n", sum);

3-4 猜数字#

C
int target = 42, guess;
while (1) {
    scanf("%d", &guess);
    if (guess == target) { printf("对了\n"); break; }
    if (guess > target)  printf("大了\n");
    else                 printf("小了\n");
}

3-5 打印金字塔#

C
int n;
scanf("%d", &n);
for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= n - i; j++) printf(" ");
    for (int j = 1; j <= 2 * i - 1; j++) printf("*");
    printf("\n");
}

第 4 章#

4-2 判断素数#

C
int is_prime(int n) {
    if (n < 2) return 0;
    for (int i = 2; i * i <= n; i++) {
        if (n % i == 0) return 0;
    }
    return 1;
}

4-5 递归求和#

C
int sum_to(int n) {
    if (n == 1) return 1;
    return n + sum_to(n - 1);
}

第 5 章#

5-3 统计数字#

C
int cnt[10] = {0};
int x;
for (int i = 0; i < 20; i++) {
    scanf("%d", &x);
    cnt[x]++;
}
for (int i = 0; i < 10; i++) {
    if (cnt[i]) printf("%d 出现了 %d 次\n", i, cnt[i]);
}

5-4 矩阵转置#

C
int a[3][4], b[4][3];

for (int i = 0; i < 3; i++)
    for (int j = 0; j < 4; j++)
        scanf("%d", &a[i][j]);

for (int i = 0; i < 3; i++)
    for (int j = 0; j < 4; j++)
        b[j][i] = a[i][j];

第 6 章#

6-2 交换两个数（指针版）#

C
void swap(int *a, int *b) {
    int t = *a;
    *a = *b;
    *b = t;
}

// 调用：swap(&x, &y);

6-4 求最大值（指针版）#

C
int max_in(int *a, int n) {
    int max = a[0];
    for (int i = 1; i < n; i++) {
        if (a[i] > max) max = a[i];
    }
    return max;
}

7-4 按成绩排序#

for (int i = 0; i < n - 1; i++) {
    for (int j = 0; j < n - 1 - i; j++) {
        if (list[j].score < list[j + 1].score) {

第 8 章#

8-4 qsort 从大到小#

C
int cmp(const void *p, const void *q) {
    int a = *(const int *)p;
    int b = *(const int *)q;
    return (a < b) - (a > b);     // 取反，变成降序
}
